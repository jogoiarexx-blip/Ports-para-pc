cmake_minimum_required(VERSION 3.22)
project(FinalFightXWeb VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "This target must be built with Emscripten (emcmake cmake).")
endif()
add_library(ffx_data src/OpenBorData.cpp)
target_include_directories(ffx_data PUBLIC src)
add_executable(FinalFightXWeb
  src/main_web.cpp
  src/Game.cpp
  src/Renderer_web.cpp
  src/Audio_web.cpp
  src/Input_web.cpp
)
target_link_libraries(FinalFightXWeb PRIVATE ffx_data)
target_include_directories(FinalFightXWeb PRIVATE src)
set_target_properties(FinalFightXWeb PROPERTIES OUTPUT_NAME "ffx" SUFFIX ".js")
target_link_options(FinalFightXWeb PRIVATE
  "-sALLOW_MEMORY_GROWTH=1"
  "-sINITIAL_MEMORY=67108864"
  "-sFORCE_FILESYSTEM=1"
  "-sDISABLE_EXCEPTION_CATCHING=0"
  "-sEXPORTED_RUNTIME_METHODS=['FS','IDBFS']"
  "-sENVIRONMENT=web"
  "-sEXIT_RUNTIME=0"
)
